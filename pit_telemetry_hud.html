const HUD_LAYOUT_KEY = 'pit_hud_layout_v1'; const panels = document.querySelectorAll('#standardHUD > div, #driveHUD #panelDrive'); function updateScale(panel) { const baseW = parseFloat(panel.dataset.baseW) || 150; const currentW = panel.offsetWidth; const scale = Math.max(0.65, Math.min(2.8, currentW / baseW)); panel.style.setProperty('--ui-scale', scale.toFixed(3)); } // レイアウトをブラウザに保存 function saveHudLayout() { const layout = {}; panels.forEach(panel => { if (!panel.id) return; layout[panel.id] = { left: panel.style.left, top: panel.style.top, width: panel.style.width, height: panel.style.height, transform: panel.style.transform, right: panel.style.right, bottom: panel.style.bottom }; }); try { localStorage.setItem(HUD_LAYOUT_KEY, JSON.stringify(layout)); } catch (e) {} } // 保存されたレイアウトを復元 function loadHudLayout() { try { const raw = localStorage.getItem(HUD_LAYOUT_KEY); if (!raw) return; const layout = JSON.parse(raw); panels.forEach(panel => { const saved = layout[panel.id]; if (!saved) return; if (saved.left) panel.style.left = saved.left; if (saved.top) panel.style.top = saved.top; if (saved.width) panel.style.width = saved.width; if (saved.height) panel.style.height = saved.height; if (saved.transform !== undefined) panel.style.transform = saved.transform; if (saved.right !== undefined) panel.style.right = saved.right; if (saved.bottom !== undefined) panel.style.bottom = saved.bottom; }); } catch (e) {} } loadHudLayout(); panels.forEach(panel => { updateScale(panel); let isDragging = false, isResizing = false; let startX, startY, startLeft, startTop, initW, initH; panel.addEventListener('mousedown', (e) => { if (e.target.classList.contains('resize-handle') || e.target.closest('#btnResetTrip')) return; isDragging = true; startX = e.clientX; startY = e.clientY; const rect = panel.getBoundingClientRect(); const parentRect = hudContainer.getBoundingClientRect(); startLeft = rect.left - parentRect.left; startTop = rect.top - parentRect.top; panel.style.transform = 'none'; panel.style.left = startLeft + 'px'; panel.style.top = startTop + 'px'; panel.style.right = 'auto'; panel.style.bottom = 'auto'; }); const handle = panel.querySelector('.resize-handle'); if (handle) { handle.addEventListener('mousedown', (e) => { e.stopPropagation(); isResizing = true; startX = e.clientX; startY = e.clientY; initW = panel.offsetWidth; initH = panel.offsetHeight; }); } window.addEventListener('mousemove', (e) => { if (isDragging) { panel.style.left = (startLeft + e.clientX - startX) + 'px'; panel.style.top = (startTop + e.clientY - startY) + 'px'; } else if (isResizing) { const minW = parseFloat(panel.dataset.minW) || 60; const minH = parseFloat(panel.dataset.minH) || 30; panel.style.width = Math.max(minW, initW + (e.clientX - startX)) + 'px'; panel.style.height = Math.max(minH, initH + (e.clientY - startY)) + 'px'; updateScale(panel); } }); window.addEventListener('mouseup', () => { if (isDragging || isResizing) { isDragging = false; isResizing = false; saveHudLayout(); } }); }); // RESETボタンで保存レイアウトを消去して初期配置に戻す document.getElementById('btnResetHud').addEventListener('click', () => { localStorage.removeItem(HUD_LAYOUT_KEY); location.reload(); });
